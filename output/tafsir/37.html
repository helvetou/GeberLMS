سورة الصافات

As-Saffat — 182 آيات

1. وَٱلصَّـٰٓفَّـٰتِ صَفّٗا

2. فَٱلزَّـٰجِرَٰتِ زَجۡرٗا

3. فَٱلتَّـٰلِيَٰتِ ذِكۡرًا

4. إِنَّ إِلَٰهَكُمۡ لَوَٰحِدٞ

5. رَّبُّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَمَا بَيۡنَهُمَا وَرَبُّ ٱلۡمَشَٰرِقِ

6. إِنَّا زَيَّنَّا ٱلسَّمَآءَ ٱلدُّنۡيَا بِزِينَةٍ ٱلۡكَوَاكِبِ

7. وَحِفۡظٗا مِّن كُلِّ شَيۡطَٰنٖ مَّارِدٖ

8. لَّا يَسَّمَّعُونَ إِلَى ٱلۡمَلَإِ ٱلۡأَعۡلَىٰ وَيُقۡذَفُونَ مِن كُلِّ جَانِبٖ

9. دُحُورٗاۖ وَلَهُمۡ عَذَابٞ وَاصِبٌ

10. إِلَّا مَنۡ خَطِفَ ٱلۡخَطۡفَةَ فَأَتۡبَعَهُۥ شِهَابٞ ثَاقِبٞ

11. فَٱسۡتَفۡتِهِمۡ أَهُمۡ أَشَدُّ خَلۡقًا أَم مَّنۡ خَلَقۡنَآۚ إِنَّا خَلَقۡنَٰهُم مِّن طِينٖ لَّازِبِۭ

12. بَلۡ عَجِبۡتَ وَيَسۡخَرُونَ

13. وَإِذَا ذُكِّرُواْ لَا يَذۡكُرُونَ

14. وَإِذَا رَأَوۡاْ ءَايَةٗ يَسۡتَسۡخِرُونَ

15. وَقَالُوٓاْ إِنۡ هَٰذَآ إِلَّا سِحۡرٞ مُّبِينٌ

16. أَءِذَا مِتۡنَا وَكُنَّا تُرَابٗا وَعِظَٰمًا أَءِنَّا لَمَبۡعُوثُونَ

17. أَوَءَابَآؤُنَا ٱلۡأَوَّلُونَ

18. قُلۡ نَعَمۡ وَأَنتُمۡ دَٰخِرُونَ

19. فَإِنَّمَا هِيَ زَجۡرَةٞ وَٰحِدَةٞ فَإِذَا هُمۡ يَنظُرُونَ

20. وَقَالُواْ يَٰوَيۡلَنَا هَٰذَا يَوۡمُ ٱلدِّينِ

21. هَٰذَا يَوۡمُ ٱلۡفَصۡلِ ٱلَّذِي كُنتُم بِهِۦ تُكَذِّبُونَ

22. ۞ٱحۡشُرُواْ ٱلَّذِينَ ظَلَمُواْ وَأَزۡوَٰجَهُمۡ وَمَا كَانُواْ يَعۡبُدُونَ

23. مِن دُونِ ٱللَّهِ فَٱهۡدُوهُمۡ إِلَىٰ صِرَٰطِ ٱلۡجَحِيمِ

24. وَقِفُوهُمۡۖ إِنَّهُم مَّسۡـُٔولُونَ

25. مَا لَكُمۡ لَا تَنَاصَرُونَ

26. بَلۡ هُمُ ٱلۡيَوۡمَ مُسۡتَسۡلِمُونَ

27. وَأَقۡبَلَ بَعۡضُهُمۡ عَلَىٰ بَعۡضٖ يَتَسَآءَلُونَ

28. قَالُوٓاْ إِنَّكُمۡ كُنتُمۡ تَأۡتُونَنَا عَنِ ٱلۡيَمِينِ

29. قَالُواْ بَل لَّمۡ تَكُونُواْ مُؤۡمِنِينَ

30. وَمَا كَانَ لَنَا عَلَيۡكُم مِّن سُلۡطَٰنِۭۖ بَلۡ كُنتُمۡ قَوۡمٗا طَٰغِينَ

31. فَحَقَّ عَلَيۡنَا قَوۡلُ رَبِّنَآۖ إِنَّا لَذَآئِقُونَ

32. فَأَغۡوَيۡنَٰكُمۡ إِنَّا كُنَّا غَٰوِينَ

33. فَإِنَّهُمۡ يَوۡمَئِذٖ فِي ٱلۡعَذَابِ مُشۡتَرِكُونَ

34. إِنَّا كَذَٰلِكَ نَفۡعَلُ بِٱلۡمُجۡرِمِينَ

35. إِنَّهُمۡ كَانُوٓاْ إِذَا قِيلَ لَهُمۡ لَآ إِلَٰهَ إِلَّا ٱللَّهُ يَسۡتَكۡبِرُونَ

36. وَيَقُولُونَ أَئِنَّا لَتَارِكُوٓاْ ءَالِهَتِنَا لِشَاعِرٖ مَّجۡنُونِۭ

37. بَلۡ جَآءَ بِٱلۡحَقِّ وَصَدَّقَ ٱلۡمُرۡسَلِينَ

38. إِنَّكُمۡ لَذَآئِقُواْ ٱلۡعَذَابِ ٱلۡأَلِيمِ

39. وَمَا تُجۡزَوۡنَ إِلَّا مَا كُنتُمۡ تَعۡمَلُونَ

40. إِلَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِينَ

41. أُوْلَـٰٓئِكَ لَهُمۡ رِزۡقٞ مَّعۡلُومٞ

42. فَوَٰكِهُ وَهُم مُّكۡرَمُونَ

43. فِي جَنَّـٰتِ ٱلنَّعِيمِ

44. عَلَىٰ سُرُرٖ مُّتَقَٰبِلِينَ

45. يُطَافُ عَلَيۡهِم بِكَأۡسٖ مِّن مَّعِينِۭ

46. بَيۡضَآءَ لَذَّةٖ لِّلشَّـٰرِبِينَ

47. لَا فِيهَا غَوۡلٞ وَلَا هُمۡ عَنۡهَا يُنزَفُونَ

48. وَعِندَهُمۡ قَٰصِرَٰتُ ٱلطَّرۡفِ عِينٞ

49. كَأَنَّهُنَّ بَيۡضٞ مَّكۡنُونٞ

50. فَأَقۡبَلَ بَعۡضُهُمۡ عَلَىٰ بَعۡضٖ يَتَسَآءَلُونَ

51. قَالَ قَآئِلٞ مِّنۡهُمۡ إِنِّي كَانَ لِي قَرِينٞ

52. يَقُولُ أَءِنَّكَ لَمِنَ ٱلۡمُصَدِّقِينَ

53. أَءِذَا مِتۡنَا وَكُنَّا تُرَابٗا وَعِظَٰمًا أَءِنَّا لَمَدِينُونَ

54. قَالَ هَلۡ أَنتُم مُّطَّلِعُونَ

55. فَٱطَّلَعَ فَرَءَاهُ فِي سَوَآءِ ٱلۡجَحِيمِ

56. قَالَ تَٱللَّهِ إِن كِدتَّ لَتُرۡدِينِ

57. وَلَوۡلَا نِعۡمَةُ رَبِّي لَكُنتُ مِنَ ٱلۡمُحۡضَرِينَ

58. أَفَمَا نَحۡنُ بِمَيِّتِينَ

59. إِلَّا مَوۡتَتَنَا ٱلۡأُولَىٰ وَمَا نَحۡنُ بِمُعَذَّبِينَ

60. إِنَّ هَٰذَا لَهُوَ ٱلۡفَوۡزُ ٱلۡعَظِيمُ

61. لِمِثۡلِ هَٰذَا فَلۡيَعۡمَلِ ٱلۡعَٰمِلُونَ

62. أَذَٰلِكَ خَيۡرٞ نُّزُلًا أَمۡ شَجَرَةُ ٱلزَّقُّومِ

63. إِنَّا جَعَلۡنَٰهَا فِتۡنَةٗ لِّلظَّـٰلِمِينَ

64. إِنَّهَا شَجَرَةٞ تَخۡرُجُ فِيٓ أَصۡلِ ٱلۡجَحِيمِ

65. طَلۡعُهَا كَأَنَّهُۥ رُءُوسُ ٱلشَّيَٰطِينِ

66. فَإِنَّهُمۡ لَأٓكِلُونَ مِنۡهَا فَمَالِـُٔونَ مِنۡهَا ٱلۡبُطُونَ

67. ثُمَّ إِنَّ لَهُمۡ عَلَيۡهَا لَشَوۡبٗا مِّنۡ حَمِيمٖ

68. ثُمَّ إِنَّ مَرۡجِعَهُمۡ لَإِلَى ٱلۡجَحِيمِ

69. إِنَّهُمۡ أَلۡفَوۡاْ ءَابَآءَهُمۡ ضَآلِّينَ

70. فَهُمۡ عَلَىٰٓ ءَاثَٰرِهِمۡ يُهۡرَعُونَ

71. وَلَقَدۡ ضَلَّ قَبۡلَهُمۡ أَكۡثَرُ ٱلۡأَوَّلِينَ

72. وَلَقَدۡ أَرۡسَلۡنَا فِيهِم مُّنذِرِينَ

73. فَٱنظُرۡ كَيۡفَ كَانَ عَٰقِبَةُ ٱلۡمُنذَرِينَ

74. إِلَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِينَ

75. وَلَقَدۡ نَادَىٰنَا نُوحٞ فَلَنِعۡمَ ٱلۡمُجِيبُونَ

76. وَنَجَّيۡنَٰهُ وَأَهۡلَهُۥ مِنَ ٱلۡكَرۡبِ ٱلۡعَظِيمِ

77. وَجَعَلۡنَا ذُرِّيَّتَهُۥ هُمُ ٱلۡبَاقِينَ

78. وَتَرَكۡنَا عَلَيۡهِ فِي ٱلۡأٓخِرِينَ

79. سَلَٰمٌ عَلَىٰ نُوحٖ فِي ٱلۡعَٰلَمِينَ

80. إِنَّا كَذَٰلِكَ نَجۡزِي ٱلۡمُحۡسِنِينَ

81. إِنَّهُۥ مِنۡ عِبَادِنَا ٱلۡمُؤۡمِنِينَ

82. ثُمَّ أَغۡرَقۡنَا ٱلۡأٓخَرِينَ

83. ۞وَإِنَّ مِن شِيعَتِهِۦ لَإِبۡرَٰهِيمَ

84. إِذۡ جَآءَ رَبَّهُۥ بِقَلۡبٖ سَلِيمٍ

85. إِذۡ قَالَ لِأَبِيهِ وَقَوۡمِهِۦ مَاذَا تَعۡبُدُونَ

86. أَئِفۡكًا ءَالِهَةٗ دُونَ ٱللَّهِ تُرِيدُونَ

87. فَمَا ظَنُّكُم بِرَبِّ ٱلۡعَٰلَمِينَ

88. فَنَظَرَ نَظۡرَةٗ فِي ٱلنُّجُومِ

89. فَقَالَ إِنِّي سَقِيمٞ

90. فَتَوَلَّوۡاْ عَنۡهُ مُدۡبِرِينَ

91. فَرَاغَ إِلَىٰٓ ءَالِهَتِهِمۡ فَقَالَ أَلَا تَأۡكُلُونَ

92. مَا لَكُمۡ لَا تَنطِقُونَ

93. فَرَاغَ عَلَيۡهِمۡ ضَرۡبَۢا بِٱلۡيَمِينِ

94. فَأَقۡبَلُوٓاْ إِلَيۡهِ يَزِفُّونَ

95. قَالَ أَتَعۡبُدُونَ مَا تَنۡحِتُونَ

96. وَٱللَّهُ خَلَقَكُمۡ وَمَا تَعۡمَلُونَ

97. قَالُواْ ٱبۡنُواْ لَهُۥ بُنۡيَٰنٗا فَأَلۡقُوهُ فِي ٱلۡجَحِيمِ

98. فَأَرَادُواْ بِهِۦ كَيۡدٗا فَجَعَلۡنَٰهُمُ ٱلۡأَسۡفَلِينَ

99. وَقَالَ إِنِّي ذَاهِبٌ إِلَىٰ رَبِّي سَيَهۡدِينِ

100. رَبِّ هَبۡ لِي مِنَ ٱلصَّـٰلِحِينَ

101. فَبَشَّرۡنَٰهُ بِغُلَٰمٍ حَلِيمٖ

102. فَلَمَّا بَلَغَ مَعَهُ ٱلسَّعۡيَ قَالَ يَٰبُنَيَّ إِنِّيٓ أَرَىٰ فِي ٱلۡمَنَامِ أَنِّيٓ أَذۡبَحُكَ فَٱنظُرۡ مَاذَا تَرَىٰۚ قَالَ يَـٰٓأَبَتِ ٱفۡعَلۡ مَا تُؤۡمَرُۖ سَتَجِدُنِيٓ إِن شَآءَ ٱللَّهُ مِنَ ٱلصَّـٰبِرِينَ

103. فَلَمَّآ أَسۡلَمَا وَتَلَّهُۥ لِلۡجَبِينِ

104. وَنَٰدَيۡنَٰهُ أَن يَـٰٓإِبۡرَٰهِيمُ

105. قَدۡ صَدَّقۡتَ ٱلرُّءۡيَآۚ إِنَّا كَذَٰلِكَ نَجۡزِي ٱلۡمُحۡسِنِينَ

106. إِنَّ هَٰذَا لَهُوَ ٱلۡبَلَـٰٓؤُاْ ٱلۡمُبِينُ

107. وَفَدَيۡنَٰهُ بِذِبۡحٍ عَظِيمٖ

108. وَتَرَكۡنَا عَلَيۡهِ فِي ٱلۡأٓخِرِينَ

109. سَلَٰمٌ عَلَىٰٓ إِبۡرَٰهِيمَ

110. كَذَٰلِكَ نَجۡزِي ٱلۡمُحۡسِنِينَ

111. إِنَّهُۥ مِنۡ عِبَادِنَا ٱلۡمُؤۡمِنِينَ

112. وَبَشَّرۡنَٰهُ بِإِسۡحَٰقَ نَبِيّٗا مِّنَ ٱلصَّـٰلِحِينَ

113. وَبَٰرَكۡنَا عَلَيۡهِ وَعَلَىٰٓ إِسۡحَٰقَۚ وَمِن ذُرِّيَّتِهِمَا مُحۡسِنٞ وَظَالِمٞ لِّنَفۡسِهِۦ مُبِينٞ

114. وَلَقَدۡ مَنَنَّا عَلَىٰ مُوسَىٰ وَهَٰرُونَ

115. وَنَجَّيۡنَٰهُمَا وَقَوۡمَهُمَا مِنَ ٱلۡكَرۡبِ ٱلۡعَظِيمِ

116. وَنَصَرۡنَٰهُمۡ فَكَانُواْ هُمُ ٱلۡغَٰلِبِينَ

117. وَءَاتَيۡنَٰهُمَا ٱلۡكِتَٰبَ ٱلۡمُسۡتَبِينَ

118. وَهَدَيۡنَٰهُمَا ٱلصِّرَٰطَ ٱلۡمُسۡتَقِيمَ

119. وَتَرَكۡنَا عَلَيۡهِمَا فِي ٱلۡأٓخِرِينَ

120. سَلَٰمٌ عَلَىٰ مُوسَىٰ وَهَٰرُونَ

121. إِنَّا كَذَٰلِكَ نَجۡزِي ٱلۡمُحۡسِنِينَ

122. إِنَّهُمَا مِنۡ عِبَادِنَا ٱلۡمُؤۡمِنِينَ

123. وَإِنَّ إِلۡيَاسَ لَمِنَ ٱلۡمُرۡسَلِينَ

124. إِذۡ قَالَ لِقَوۡمِهِۦٓ أَلَا تَتَّقُونَ

125. أَتَدۡعُونَ بَعۡلٗا وَتَذَرُونَ أَحۡسَنَ ٱلۡخَٰلِقِينَ

126. ٱللَّهَ رَبَّكُمۡ وَرَبَّ ءَابَآئِكُمُ ٱلۡأَوَّلِينَ

127. فَكَذَّبُوهُ فَإِنَّهُمۡ لَمُحۡضَرُونَ

128. إِلَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِينَ

129. وَتَرَكۡنَا عَلَيۡهِ فِي ٱلۡأٓخِرِينَ

130. سَلَٰمٌ عَلَىٰٓ إِلۡ يَاسِينَ

131. إِنَّا كَذَٰلِكَ نَجۡزِي ٱلۡمُحۡسِنِينَ

132. إِنَّهُۥ مِنۡ عِبَادِنَا ٱلۡمُؤۡمِنِينَ

133. وَإِنَّ لُوطٗا لَّمِنَ ٱلۡمُرۡسَلِينَ

134. إِذۡ نَجَّيۡنَٰهُ وَأَهۡلَهُۥٓ أَجۡمَعِينَ

135. إِلَّا عَجُوزٗا فِي ٱلۡغَٰبِرِينَ

136. ثُمَّ دَمَّرۡنَا ٱلۡأٓخَرِينَ

137. وَإِنَّكُمۡ لَتَمُرُّونَ عَلَيۡهِم مُّصۡبِحِينَ

138. وَبِٱلَّيۡلِۚ أَفَلَا تَعۡقِلُونَ

139. وَإِنَّ يُونُسَ لَمِنَ ٱلۡمُرۡسَلِينَ

140. إِذۡ أَبَقَ إِلَى ٱلۡفُلۡكِ ٱلۡمَشۡحُونِ

إذا سمعوا بكلمة التوحيد استكبروا وأبوا إلا الشرك .

36- وَيَقُولُونَ أَبِنَّا لَتَارِكُوا عَالِهَتِنَا [أي لِمَ نترك عبادة أصنامنا؟! وهو

استفهام بمعنى النفي ] لِشَاعِرِ تَجْنُونِ ﴾ [أي لقول رجل يأتي بكلام

منظوم لا يعقل ما يقول لأنه يخبرنا بالبعث بعد الموت]، يعنون محمدًا

عليه الصلاة والسلام.

جَنَّتِ النَّعِيمِ عَلَى سُرُرٍ مُّتَقَبِلِينَ يُطَافُ عَلَيْهِم بِكَأْسٍ مِّن مَّعِينٍ )

37- بَلْ جَاءَ بِالْحَقِّ ردُّ على المشركين، [أي: ليس بشاعر ولا

مجنون، بل هو رسول جاء بما يوافق العقول السليمة] وَصَدَقَ

الْمُرْسَلِينَ ) [أي: وحقَّقَ بشارات الأنبياء الماضين به، ووافق ما

جاؤوا به من توحيد الله وطاعته ] كقوله : مُصَدِّقًا لِمَا بَيْنَ يَدَيْهِ

[سورة البقرة].

38 - إِنَّكُمْ لَذَائِقُوا الْعَذَابِ الْأَلِيمِ بسبب الإشراك وتكذيب

الرسل، إذ هو ثمرة عملكم].

-39- وَمَا تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ﴾ [إلا جزاء ما كنتم تعملون]

بلا زيادة .

40 - إِلَّا عِبَادَ اللَّهِ الْمُخْلَصِينَ لكن عباد الله الذين أخلصهم الله

تعالى لطاعته ودينه وولايته .

41 - أُوْلَئِكَ لَهُمْ رِزْقٌ مَّعْلُومٌ منعوت بخصائص خُلِقَ عليها من

طيب طعم ورائحة ولَذَّةٍ وحُسنِ منظر .

42 - فَوَاكِهُ فسَّرَ الرزق المعلوم بالفواكه، وهي كلُّ ما يُتلذذ به،

[وهو إشارة إلى أنهم يتناولون المأكولات التي يُتنعم بها، وهو كقوله :

وَلَحْمِ طَيْرِ مِّمَّا يَشْتَهُونَ ﴾ [سورة الواقعة] أي يأكلونها مشتهين لها، لا

مضطرين إليها ] وَهُم مُّكْرَمُونَ ﴾ مُعَظَّمون .

43 - فِي جَنَّاتِ النَّعِيمِ ﴾ [في جنات ليس فيها إلا النعيم].

44- عَلَى سُرُرٍ مُّتَقَبِلِينَ ﴾ التقابلُ أَتَمُّ للسرور وءانَسُ .

45 - يُطَافُ عَلَيْهِم بِكَأْسٍ يقال للزجاجة فيها الخمر كأس، وتُسمَّى

الخَمْرُ نفسها كأسًا مِن مَّعِينِ من شراب مَعِين، أو من نهر معين،

عِينُ كَأَنَّهُنَّ بَيْضُ مَكْنُونٌ فَأَقْبَلَ بَعْضُهُمْ عَلَى بَعْضٍ يَتَسَاءَلُونَ )

وهو الجاري على وجه الأرض الظاهرُ للعيون، وُصف بما وصف به

الماء لأنه يجري في الجنة في أنهار كما يجري الماء، قال الله تعالى :

وَأَنْهَرُ مِنْ خَمْرٍ ﴾ [سورة محمد].

46 - بَيْضَاءَ صفة للكأس لَذَّةٍ وُصفت باللذة كأنها نفس اللذة

وعينها ، أو ذات لذة لِلشَّارِبِينَ الشاربيها، ليست كخمور الدنيا

في كراهة الطعم].

47 - لَا فِيهَا غَوْلُ لا تغتال عقولهم كخمور الدنيا، وهو من غالَهُ :

إذا أَهْلَكَهُ وأفسده وَلَا هُمْ عَنْهَا يُنزَفُونَ يَسكَرون، مِن نُزِفَ

الشارب إذا ذهب عقله .

48 - وَعِندَهُمْ قَصِرَاتُ الطَّرْفِ قَصَرْنَ أبصارهن على أزواجهن لا

يَمْدُدْنَ طَرْفًا إلى غيرهم عين جمع عيناءَ أي نجلاء(نجلاء واسعة شَقِّ العين مع حسن.) واسعة

العين .

49 - كَأَنَّهُنَّ بَيْضُ مَكْنُونٌ مَصُونٌ ، شَبَّهَهُنَّ بِبَيضِ النَّعام المكنون

[تَكُنُّهُ النعامة بالريش؛ أي تستره وتصونه عن الغبار ونحوه] - في

الصفاء والبياض المخلوط بأدنى صفرة، وبها تُشبّه العرب النساء

وتُسَمِّيهن بيضات الخدور (الخدور جمعُ الخِدْرِ : وهو ستر يُمد للجارية في ناحية البيت.).

50 - فَأَقْبَلَ بَعْضُهُمْ يعني أهل الجنة عَلَى بَعْضٍ يَتَسَاءَلُونَ

يشربون ويتحادثون بما أنعم الله عليهم من حين كانوا في الدنيا إلى أن

صاروا إلى الجنة .

قَالَ قَابِلُ مِنْهُمْ إِنِّي كَانَ لِي قَرِينٌ يَقُولُ أَعِنَّكَ لَمِنَ الْمُصَدِّقِينَ أَعْذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَامًا

أَعِنَّا لَمَدِينُونَ قَالَ هَلْ أَنتُم مُّطَّلِعُونَ فَاطَّلَعَ فَرَءَاهُ فِي سَوَاءِ الْجَحِيمِ قَالَ تَاللَّهِ إِن كِدت

لَتُرْدِينِ وَلَوْلَا نِعْمَةُ رَبِّي لَكُنتُ مِنَ الْمُحْضَرِينَ أَفَمَا نَحْنُ بِمَيِّتِينَ إِلَّا مَوْنَتَنَا الْأُولَى وَمَا

نَحْنُ بِمُعَذِّبِينَ

51 - قَالَ قَابِلُ مِنْهُمْ إِنِّي كَانَ لِي قَرِينٌ [صاحب مُقارن كافر

بالبعث ] .

52 - يَقُولُ أَعِنَّكَ لَمِنَ الْمُصَدِّقِينَ * بيوم الدين؟! [استفهام بمعنى

الإنكار].

53 - أَءِ ذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَامًا أَعِنَّا لَمَدِينُونَ لَمَجْزِيون

[ومُحاسَبون].

54- قَالَ ذلك القائل لرفقائِهِ في الجنة الذين كان وإياهم

يتساءلون ] : هَلْ أَنتُم مُّطَّلِعُونَ إلى النار لأريكم ذلك القرين . قيل :

إن في الجنة كُوّى ينظر أهلها منها إلى أهل النار.

هه - فَاطَّلَعَ المسلمُ فَرَاهُ أي [رأى قــريــنــه فِي سَوَاءِ

الْجَحِيمِ في وسطها .

56 - قَالَ تَاللَّهِ إِن كِدتَّ لَتُرْدِينِ ) الإرداء : الإهلاك، أي لقد كنت

قاربت أن تهلكني بإضلالك ، أو ما أردت إلا أن تهلكني].

57 - وَلَوْلَا نِعْمَةُ رَبِّي وهي العِصمةُ والتوفيق في الاستمساك بعُروة

الإسلام لَكُنتُ مِنَ الْمُحْضَرِينَ من الذين أحضروا العذاب كما

أحضرته أنت وأمثالك .

58 - 59- أَفَمَا نَحْنُ بِمَيِّتِينَ إِلَّا مَوْنَتَنَا الْأُولَى وَمَا نَحْنُ بِمُعَذِّبِينَ ﴾

أنحن مُخلَّدون منعمون فما نحن بميتين ولا مُعَذَّبين، والمعنى أن هذه

حال المؤمنين، وهو ألّا يذوقوا إلا المَوتَةَ الأولى التي كانت في

الدنيا، وهي متناولةٌ لِمَا في القبر بعد الإحياء للسؤال] بخلاف الكفار،

فإنهم فيما يتمنون فيه الموت كل ساعة. وقيل لحكيم: ما شَرٌّ مِنَ

الموت؟ قال : الذي يُتمنَّى فيه الموت. وهذا قول يقوله المؤمن تحدثًا

بنعمة الله بمسمع من قرينه ليكون توبيخا له وزيادة تعذيب، [وهو كقول

الشخص عند التعجب : أَوَكلُّ هذه النعم لي ؟! لا يكون إنكارًا، بل

تعجبًا وشكرا ] .

60 - إِنَّ هَذَا الأمر الذي نحن فيه هُوَ الْفَوْزُ الْعَظِيمُ )

ثم قال الله عز وجل :

61- لِمِثْلِ هَذَا فَلْيَعْمَلِ الْعَامِلُونَ ﴾ النيل مثل هذا يجب أن يعمل

العاملون، لا للحظوظ الدنيوية المشوبة بالآلام، السريعة الانصرام]،

وقيل : هو أيضًا من كلام المؤمن المتحدث بنعمة الله في الجنة].

62 - أَذَلِكَ خَيْرُ نُزُلًا أَمْ شَجَرَةُ الزَّقُومِ ) أي نعيم الجنة وما فيها

من اللذات والطعام والشراب خير نزلا أم شجرة الزقوم خير نزلا؟

والنُّزل ما يُقام للنازل بالمكان من ضيف وغيره من الرزق. والزقوم:

شجرةٌ مُرَّةٌ تكون بتهامة، من أخبث الشجر، ينبتها الله تعالى في

الجحيم].

63 - إِنَّا جَعَلْنَهَا فِتْنَةً لِلظَّالِمِينَ ﴾ مِحنةً وعذابًا لهم في الآخرة،

أو ابتلاء لهم في الدنيا، وذلك أنهم قالوا : كيف يكون في النار

شجرة، والنار تحرق الشجر ؟ فكذبوا .

64 - إِنَّهَا شَجَرَةٌ تَخْرُجُ فِي أَصْلِ الْجَحِيمِ قيل : مَنبتها في قعر

جهنم، وأغصانها ترتفع إلى دركاتها .

65 - طَلْعُهَا كَأَنَّهُ رُءُوسُ الشَّيَاطِينِ الطلعُ للنخلة فاستعير لِمَا طَلَعَ

الجزء الثاني

من شجرة الزقوم من حَمْلِها، وشُبِّهَ برؤوس الشياطين للدلالة على تناهيه

في الكراهة وقبح المنظر، لأن الشيطان مكروه مستقبح في طباع الناس

لاعتقادهم أنه شَرِّ مَحْضُ .

66 - فَإِنَّهُمْ لَاكِلُونَ مِنْهَا من الشجرة، أي من طلعها فَمَالِئُونَ مِنْهَا

الْبُطُونَ فمالئون بطونهم لما يغلبهم من الجوع الشديد.

67 - ثُمَّ إِنَّ لَهُمْ عَلَيْهَا على أكلها لَشَوْبًا لَخَلْطًا وَلَمِرَاجًا مِنْ

حَمِيمٍ ماء حار يشوي وجوههم ويُقَطِع أمعاءهم، والمعنى: ثم

إنهم يملؤون البطون من شجر الزقوم، وهو حار يحرق بطونهم

ويُعطشهم فلا يُسقون إلا بعد مَلِي (أي : زمان طويل.) تعذيبًا لهم بذلك العطش، ثم

يُسقون ما هو أحَرُّ، وهو الشراب المشوب بالحميم، [قيل : يُخلط لهم

الحميم بغساق أعينهم وصديد من قيحهم ودمائهم، وقيل : يُخرج لهم

الزقوم بالحميم ليُجمع لهم بين مرارة الزقوم وحرارة الحميم].

68 - ثُمَّ إِنَّ مَرْجِعَهُمْ لَإِلَى الْجَحِيمِ أي أنهم يُذهب بهم عن

مقارهم ومنازلهم في الجحيم، وهي الدَّرَكات التي أُسكنوها إلى شجرة

الزقوم، فيأكلون إلى أن يمتلؤوا، ويُسقون بعد ذلك، ثم يُرجعون إلى

دركاتهم .

69 - 70 - إِنَّهُمْ أَلْفَوْا ءَابَاءَهُمْ ضَالِينَ ﴾ [وجدوا ءاباءهـم عـلـى

ضلال] فَهُمْ عَلَى ءَاثَرِهِمْ يُهْرَعُونَ ﴾ عَلَّلَ استحقاقهم للوقوع في تلك

الشدائد بتقليد الآباء في الدين واتّباعهم إياهم في الضلال وترك اتباع

الدليل والإهراعُ : الإسراعُ الشديد كأنهم يُحَثُونَ حَقًّا .

وَأَهْلَهُ مِنَ الْكَرْبِ الْعَظِيمِ وَجَعَلْنَا ذُرِّيَّتَهُ هُمُ الْبَاقِينَ وَتَرَكْنَا عَلَيْهِ فِي الْآخِرِينَ )

71- وَلَقَدْ ضَلَّ قَبْلَهُمْ قبل قوم قريش أَكْثَرُ الْأَوَّلِينَ ) يعني

الأمم الخالية بالتقليد وترك النظر والتأمل.

72- وَلَقَدْ أَرْسَلْنَا فِيهِم مُنذِرِينَ أنبياء حَذَّروهم العواقب.

73- فَانظُرْ كَيْفَ كَانَ عَاقِبَةُ الْمُنذَرِينَ الذين أُنذِروا وحذروا

[فلم يقبلوا وعاندوا]؛ أي أهلكوا جميعًا.

74 - إِلَّا عِبَادَ اللَّهِ الْمُخْلَصِينَ إلا الذين ءامنوا منهم وأخلصهم

الله لدينه .

ولما ذكر إرسال المنذرين في الأمم الخالية وسوء عاقبة المنذرين

[المعاندين] أتبع ذلك ذكر نوح ودعائه إياه حين أَيسَ من قومه

بقوله :

أَنِّي مَغْلُوبٌ فَانتَصِرُ ﴾ [سورة القمر ] فَلَنِعْمَ الْمُجِيبُونَ تقديره :

ولقد نادانا نوح فوالله لنعم المجيبونَ نحنُ ، والمعنى أنا أجبناه أحسن

76 - وَنَجَيْنَهُ وَأَهْلَهُ ومَنْ ءامن به وأولاده مِنَ الْكَرْبِ

وهو غَمُ الغَرَقِ ، [ أو الغَمُّ الذي كان فيه من أذى قومه].

العظيم

الناس كلهم من ذرية نوح، وكان لنوح ثلاثة أولاد: سام وهو أبو

178 - 179 - وَتَوَلَّ عَنْهُمْ حَتَّى حِينٍ وَأَبْصِرْ فَسَوْفَ يُبْصِرُونَ )

وإنما ثُنّي ليكون تسلية (أي : تخفيفا للحزن.) على تسلية، وتأكيدًا لوقوع الميعاد إلى تأكيد،

المخلوقات إلا وهو ربُّها ومالكها، كقوله تعالى : وَتُعِزُّ مَن تَشَاءُ

181 - وَسَلَامُ عَلَى الْمُرْسَلِينَ ﴾ عَمَّ الرُّسُلَ بالسلام بعد ما خَصَّ

البعض في السورة، لأن في تخصيص كلّ بالذكر تطويلا.

182 - وَالْحَمْدُ لِلَّهِ رَبِّ الْعَالَمِينَ على هلاك الأعداء ونُصرة

الأنبياء. اشتملت السورة على ذِكرِ ما قاله المشركون في الله ونسبوه إليه

مما هو منزّه عنه، وما عاناه المرسلون من جهتهم وما خُوِّلُوه (أي : أعطوه.) في
